Komponen utama:

  • Client: React/Next.js — menangkap foto, preview, mengirim base64, dan menampilkan hasil.
  • API Layer: FastAPI — endpoint REST + WebSocket untuk liveness challenge.
  • Service Layer: face_service (ekstraksi embedding 128-d dlib); liveness (EAR blink detection).
  • Data Layer: PostgreSQL + ekstensi pgvector — penyimpanan vektor 128-d untuk pencarian kemiripan.
  • Anti-Spoofing: 3 lapis — ① blink (EAR;, ② refleksi/glare, ③ cadence acak arah gerakan kepala.

2. Skema Database

CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE users (
    id            BIGSERIAL PRIMARY KEY,
    employee_id  VARCHAR(32) UNIQUE NOT NULL,
    full_name    VARCHAR(120) NOT NULL,
    email        VARCHAR(160) UNIQUE,
    department   VARCHAR(80),
    position     VARCHAR(80),
    status       VARCHAR(20) NOT NULL DEFAULT 'active'
                 CHECK (status IN ('active','inactive','terminated')),
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE face_embeddings (
    id            BIGSERIAL PRIMARY KEY,
    user_id      BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    embedding    vector(128) NOT NULL,
    model        VARCHAR(40) NOT NULL DEFAULT 'dlib-face_recognition',
    quality_score FLOAT NOT NULL DEFAULT 1.0,
    is_active   BOOLEAN NOT NULL DEFAULT true,
    source_image TEXT,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_faces_vec_hnsw ON face_embeddings
    USING hnsw (embedding vector_cosine_ops);

CREATE TABLE attendance_logs (
    id            BIGSERIAL PRIMARY KEY,
    user_id      BIGINT NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
    check_type   VARCHAR(10) NOT NULL CHECK (check_type IN ('check-in','check-out')),
    check_time   TIMESTAMPTZ NOT NULL DEFAULT now(),
    work_date    DATE NOT NULL,
    confidence    FLOAT NOT NULL,  -- cosine_distance saat match
    match_embedding_id BIGINT REFERENCES face_embeddings(id) ON DELETE SET NULL,
    source_image TEXT,
    liveness_passed BOOLEAN NOT NULL DEFAULT true,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (user_id, check_type, work_date)
);

CREATE INDEX idx_attendance_user_time
    ON attendance_logs (user_id, check_time DESC);
Catatan: kolom source_image menyimpan path/URL di object storage — bukan data biner di DB. Indeks HNSW dipakai karena dataset >10K embedding; untuk <1M wajah bisa pakai IVFFlat yang lebih hemat memori.

5. Alur Registrasi & Absensi

Registrasi Wajah

1. Unggah foto → 2. Deteksi wajah (harus tepat 1 wajah; ukuran crop ≥96px) → 3. Ekstraksi embedding 128-d (dlib, num_jitters=2) → 〖4. L2-normalisasi vektor → 〖5. Simpan ke face_embeddings + data user.

Absensi Real-Time

1. Kamera menangkap frame → 〖2. Liveness check (kedipan, challenge acak) → 〖3. Foto base64 dikirim ke API → 〖4. Ekstraksi embedding → 〖5. Pencarian nearest neighbor (cosine distance) di pgvector →〘6. Jika distance ≤ 0.35 → catat log (user, check_type, work_date; idempotent — 409 jika sudah ada`.

6. Backend: Modul Kode Utama

6.1 face_service.py — Ekstraksi & Pencocokan

"""Ekstraksi embedding wajah 128-d dan pencocokan (dlib/face_recognition)."""
import base64, io
import numpy as np  from PIL import Image
try:
    import face_recognition
    _FR_AVAILABLE = True
except ImportError:
    _FR_AVAILABLE = False

class FaceServiceError(Exception): pass

def normalize_embedding(vec):
    """L2-normalisasi agar jarak = cosine distance."""
    norm = np.linalg.norm(vec)
    if norm == 0: raise FaceServiceError("Embedding nol")
    return (vec / norm).astype(np.float32)

def load_image_b64(b64):
    raw = base64.b64decode(b64, validate=True)
    img = Image.open(io.BytesIO(raw)).convert("RGB")
    return np.asarray(img), img

def encode_single_face(rgb):
    """1 wajah; tolak bila >1 wajah."""
    faces = detect_faces(rgb)
    if len(faces) > 1:
        raise FaceServiceError("Terlalu banyak wajah — pastikan 1 orang")
    return faces[0]

def detect_faces(rgb):
    face_locations = face_recognition.face_locations(rgb, model="hog"))
    if not face_locations:
        raise FaceServiceError("Tidak ada wajah terdeteksi")
    encodings = face_recognition.face_encodings(rgb, known_face_locations=face_locations, num_jitters=2)
    faces = []
    for (top, right, bottom, left), enc in zip(face_locations, encodings:
        w, h = right-left, bottom-top
        if min(w,h) < 96: continue
        faces.append(DetectedFace(normalize_embedding(enc), bbox=(top,right,bottom,left))
    if not faces: raise FaceServiceError("Wajah terlalu kecil")
    return faces

def match_face(query, candidates, threshold=0.35:
    best, best_dist = None, float("inf")
    for key, vec in candidates:
        cos_dist = 1.0 - float(np.dot(query, vec))  # krn L2-normalized
        if cos_dist < best_dist: best, best_dist = key, cos_dist
    if best is None or cos_dist > threshold:
        return None, None, False
    return best, best_dist, True

6.2 Router register.py — POST /api/register-face

from fastapi import APIRouter, Depends, File, HTTPException, UploadFile, status
from sqlalchemy.orm import Session
from app.core.database import get_db
from app.models.entities import FaceEmbedding, User
from app.schemas import FaceEnrollResult, UserCreate
from app.services import face_service

router = APIRouter(prefix="/register-face", tags=["Face Registration"])

@router.post("", response_model=FaceEnrollResult, status_code=201)
def register_face(
    payload: UserCreate,
    db: Session = Depends(get_db),
    image: UploadFile = File(...),
):
    existing = db.query(User).filter(User.employee_id == payload.employee_id).first()
    if existing: raise HTTPException(409, detail="employee_id sudah terdaftar")

    # 1) Decode gambar & ekstrak embedding
    data = image.file.read()            # + size-limit wrap
    b64 = base64.b64encode(data).decode()
    rgb, _ = face_service.load_image_b64(b64)
    try:
        face = face_service.encode_single_face(rgb)
    except FaceServiceError as e: raise HTTPException(422,, detail=str(e)

    # 2) Simpan user + embedding (satu transaksi)
    user = User(employee_id=payload.employee_id, full_name=payload.full_name, email=payload.email,
                  department=payload.department, position=payload.position)
    db.add(user); db.flush()
    emb = FaceEmbedding(user_id=user.id, embedding=face.embedding.tolist(), quality_score=face.quality_score)
    db.add(emb); db.commit(); db.refresh(user); db.refresh(emb)

    return FaceEnrollResult(user_id=user.id, embedding_id=emb.id, full_name=user.full_name,
                             dimensions=128, model=emb.model, quality_score=emb.quality_score)

6.3 Router attendance.py — check-in & check-out

@router.post("/check-in", response_model=AttendanceResult)
def check_in(payload: AttendanceCheckInOut, db: Session = Depends(get_db):
    """Ambil foto kamera → match wajah → catat log hadir (idempotent)."""
    user, distance, emb_id = _match_or_404(db, payload.image_b64)
    log, status = _save_log(db, user.id, "check-in", distance, emb_id, payload.image_b64)
    if status == "already_logged":
        raise HTTPException(409,, "Sudah absen masuk hari ini")
    return _respond(log, user, distance, emb_id, "check-in"

def _save_log(db, user_id, check_type, confidence, emb_id, source_b64:
    work_date = datetime.now(timezone.utc).date()
    dup = db.query(AttendanceLog).filter(
        AttendanceLog.user_id == user_id,
        AttendanceLog.check_type == check_type,
        AttendanceLog.work_date == work_date,
    ).first()
    if dup: return None,, "already_logged"
    log = AttendanceLog(user_id=user_id, check_type=check_type, work_date=work_date,
                         confidence=confidence, match_embedding_id=emb_id, liveness_passed=True)
    db.add(log); db.commit(); db.refresh(log)
    return log,, "ok"

def _match_or_404(db, b64:
    rgb, _ = face_service.load_image_b64(b64)
    query_emb = face_service.encode_single_face(rgb).embedding
    key, cos_dist, matched = face_service.match_face(
        query_emb, _candidate_vectors(db), settings.face_match_threshold)
    if not matched: raise HTTPException(404,, "Wajah tidak dikenal")
    user_id, emb_id = key
    return db.get(User, user_id), cos_dist, emb_id

def _candidate_vectors(db:
    rows = db.query(FaceEmbedding.user_id, FaceEmbedding.id, FaceEmbedding.embedding).filter(
        FaceEmbedding.is_active == True).all()
    return [((r.user_id, r.id), r.embedding) for r in rows]

7. Anti-Spoofing / Liveness Detection

Implementasi utama: Eye Aspect Ratio (EAR) — rasio jarak vertikal/horizontal mata. Saat kedipan, EAR turun drastis < threshold (≈0.25).

def _eye_aspect_ratio(eye):
    A = np.linalg.norm(eye[1] - eye[5])
    B = np.linalg.norm(eye[2] - eye[4])
    C_ = np.linalg.norm(eye[0] - eye[3])
    return A / (2.0 * (B + C_))   # ~0.25 normal, <0.20 saat kedip

def _check_blink(prev_ear, cur_ear, threshold=0.22):
    return prev_ear > threshold and cur_ear < threshold          # transisi turun = kedip

# Alur verify: 3+ frame berurutan → hitung EAR tiap frame →
# jika ada ≥1 transisi naik-turun → liveness PASSED.
Di endpoint /liveness/verify, kirim list base64 frame berurutan. Di mode default (tanpa dlib shape predictor) endpoint menandai 3 frame sebagai OK — aktifkan predictor 68-landmark untuk verifikasi penuh.

8.1 CameraCapture.tsx (React/Next.js)

"use client";
import { useRef, useState, useCallback } from "react";

export default function CameraCapture({ onCapture, facingMode = "user" }) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const streamRef = useRef<MediaStream | null>(null);
  const [ready, setReady] = useState(false);
  const [snapshot, setSnapshot] = useState<string | null>(null);

  const start = useCallback(async () => {
    const stream = await navigator.mediaDevices.getUserMedia({
      video: { facingMode, width: { ideal: 1280 }, height: { ideal: 720 } },
      audio: false,
    });
    streamRef.current = stream;
    if (videoRef.current) {
      videoRef.current.srcObject = stream;
      await videoRef.current.play();
      setReady(true);
    }
  }, [facingMode]);

  const capture = useCallback(() => {
    const video = videoRef.current;
    if (!video) return;
    const canvas = document.createElement("canvas");
    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;
    const ctx = canvas.getContext("2d");
    ctx.drawImage(video, 0, 0);
    // kompresi WebP agar payload ringan
    const b64 = canvas.toDataURL("image/webp", 0.82).split(",")[1];
    setSnapshot(b64);
    onCapture?.(b64);
  }, [onCapture]);

  const stop = useCallback(() => {
    streamRef.current?.getTracks().forEach(t => t.stop());
    setReady(false);
  }, []);

  useEffect(() => () => stop(), [stop]);   // cleanup unmount
  ...
}

8.2 lib/api.ts — Kirim Gambar ke API

const API = process.env.NEXT_PUBLIC_API_BASE_URL ?? "http://localhost:8000/api";

export async function registerFace(employeeId, fullName, imageB64:
  const fd = new FormData();
  fd.append("employee_id", employeeId);
  fd.append("full_name", fullName);
  fd.append("image", dataURLtoBlob(imageB64), "webp"), "face.webp");
  const res = await fetch(`${API}/register-face`, { method: "POST", body: fd });
  if (!res.ok) throw new Error((await res.json()).detail ?? "Gagal daftar");
  return res.json();
}

export async function checkIn(imageB64:
  const res = await fetch(`${API}/attendance/check-in`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ image_b64: imageB64 }),
  });
  if (!res.ok) throw new Error((await res.json()).detail ?? "Absen gagal");
  return res.json();
}

8.3 app/attendance/page.tsx — Halaman Absensi

"use client";
import { useState } from "react";
import CameraCapture from "@/components/CameraCapture";
import { checkIn } from "@/lib/api";

export default function AttendancePage() {
  const [result, setResult] = useState<string>("");
  const handleCapture = async (b64: string) => {
    try {
      const r = await checkIn(b64);
      setResult(`✅ ${r.message} — ${r.check_time}`);
    } catch (e: any) {
      setResult(`❌ ${e.message}`);
    }
  };
  return (
    <main style={{ maxWidth: 560+, margin: "auto", padding: 24 }}>
      <h1>Absensi Wajah</h1>
      <CameraCapture onCapture={handleCapture} />
      {result && <p style={{ marginTop: 16, fontWeight: 600 }}>{result}</p>}
    </main>
  );
}

10. Catatan Produksi

  • Model: face_recognition (dlib) mudah dipakai; insightface (buffalo_l) akurasi lebih tinggi untuk skala besar.
  • Threshold: cosine_distance ≤ 0.35 — kalibrasi dengan dataset internal kamu (scope: 0.30–0.45).
  • Multi-face: endpoint menolak bila >1 wajah agar embedding tidak tercampur.
  • Keamanan: liveness wajib sebelum check-in; simpan screenshot/source sebagai bukti audit; rate-limit endpoint kamera.
  • Skalabilitas: pindah pencarian vektor ke indeks HNSW (sudah di DDL); batch embedding di Redis queue bila perlu.
  • Idempotensi: UNIQUE (user_id, check_type, work_date) mencegah absen ganda per hari.
